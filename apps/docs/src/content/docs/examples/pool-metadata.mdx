import { Tabs, TabItem } from '@astrojs/starlight/components';

Reads a Uniswap V3 pool, its two tokens, their symbols, a `decimals` value with a default and a
holder's balance in one `eth_call`.

## Program

<Tabs syncKey="example">
  <TabItem label="script.ts">
    ```ts title="script.ts" {13-14, 16-17, 30}
    import { evscript, t } from '@maxencerb/evs';
    import { erc20Abi } from 'viem';
    import { poolAbi } from './abi';

    export const poolMeta = evscript(
      { name: 'poolMeta', args: [t.address, t.address] },
      (s, pool, holder) => {
        const token0 = s.read({ address: pool, abi: poolAbi, functionName: 'token0' });
        const token1 = s.read({ address: pool, abi: poolAbi, functionName: 'token1' });
        const fee = s.read({ address: pool, abi: poolAbi, functionName: 'fee' });
        const slot0 = s.read({ address: pool, abi: poolAbi, functionName: 'slot0' });

        const symbol0 = s.read({ address: token0, abi: erc20Abi, functionName: 'symbol' });
        const symbol1 = s.read({ address: token1, abi: erc20Abi, functionName: 'symbol' });

        const d = s.tryRead({ address: token0, abi: erc20Abi, functionName: 'decimals' });
        const decimals0 = s.select(d.success, d.value, 18);

        const balance0 = s.read({
          address: token0,
          abi: erc20Abi,
          functionName: 'balanceOf',
          args: [holder],
        });

        return s.return({
          token0,
          token1,
          fee,
          tick: slot0[1],
          symbol0,
          symbol1,
          decimals0,
          balance0,
        });
      },
    );
    ```
  </TabItem>
  <TabItem label="abi.ts">
    ```ts title="abi.ts"
    import { parseAbi } from 'viem';

    export const poolAbi = parseAbi([
      'function token0() view returns (address)',
      'function token1() view returns (address)',
      'function fee() view returns (uint24)',
      // slot0 returns sqrtPriceX96, tick, then five values this example does not read
      'function slot0() view returns (uint160, int24, uint16, uint16, uint16, uint8, bool)',
    ]);
    ```
  </TabItem>
  <TabItem label="client.ts">
    ```ts title="client.ts" {8}
    import { createPublicClient, http } from 'viem';
    import { mainnet } from 'viem/chains';
    import { poolMeta } from './script';

    const client = createPublicClient({ chain: mainnet, transport: http() });

    const result = await client.readContract({
      ...poolMeta.compile().toViem(),
      functionName: 'poolMeta',
      args: [
        '0x88e6A0c2dDD26FEEb64F039a2c41296FcB3f5640', // USDC/WETH 0.05% pool
        '0xd8dA6BF26964aF9D7eEd9e03E53415D37aA96045', // holder
      ],
    });
    ```

## Result

viem infers `result` from the script:

```ts
declare const result: {
  token0: `0x${string}`; // USDC
  token1: `0x${string}`; // WETH
  fee: number; // 500
  tick: number;
  symbol0: string; // 'USDC'
  symbol1: string; // 'WETH'
  decimals0: number; // 6
  balance0: bigint;
};
```

## What to notice

- **A read feeds the next.** `token0` is an `Expr<'address'>`, a value that exists only inside
  the call, and [`s.read`](/reference/builder/read/) takes it as the `address` of the token reads:
  the dependency a [multicall cannot batch](/getting-started/why-evs/).
- **Several outputs, one tuple.** `slot0()` returns seven values, so `s.read` returns a tuple of
  `Expr`s and `slot0[1]` is the tick.
- **A default, not a revert.** [`s.tryRead`](/reference/builder/try-read/) reports a failed or
  undecodable call in `success`, and [`s.select`](/reference/builder/select/) falls back to `18`.
- **Typed in, typed out.** The `args` of [`evscript`](/reference/evscript/) type `pool` and
  `holder`, and the keys of [`s.return`](/reference/builder/return/) are the fields of `result`.
- **No ABI for the script.** [`compile()`](/reference/compile/) once and spread
  [`toViem()`](/reference/artifact/to-viem/) into `readContract`: it carries the `abi` and the
  bytecode, and nothing is deployed.

The repository runs this example against a local anvil:
[examples/pool-meta](https://github.com/maxencerb/evs/tree/main/examples/pool-meta).
