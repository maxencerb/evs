import { Card, CardGrid } from '@astrojs/starlight/components';

```ts twoslash
import { evscript, t, compile } from '@maxencerb/evs';
import { erc20Abi } from 'viem';
// ---cut-start---
const poolAbi = [
  {
    type: 'function',
    name: 'token0',
    stateMutability: 'view',
    inputs: [],
    outputs: [{ name: '', type: 'address' }],
  },
] as const;
// ---cut-end---
// Two DEPENDENT reads, one eth_call: the second call's target IS the first
// call's result — a chain multicall can't batch (it needs every address up front).
const poolToken0 = evscript({ name: 'poolToken0', args: [t.address] }, (s, pool) => {
  const token0 = s.call({ address: pool, abi: poolAbi, functionName: 'token0' });
  //    ^?
  const symbol = s.call({ address: token0, abi: erc20Abi, functionName: 'symbol' });
  return s.return({ token0, symbol });
});

const script = compile(poolToken0);
// → runtime bytecode + a literal-typed ABI viem infers your args & result from
```
